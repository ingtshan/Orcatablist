### R4 索引源抽象（`src/indexer.ts` + 新 `src/sources/`）
保持 KISS，不要过度 OOP。定义一个轻量 source 描述：
```ts
export interface SessionFileInfo { agent: Agent; sid: string; path: string; size: number; mtime: number; }
export interface SessionSource {
  agent: Agent;
  discover(): SessionFileInfo[];                       // 发现该 agent 的所有会话文件
  parseLine(line: string): ParsedEvent;                // 复用/特化解析
  prepare?(): void;                                    // 每轮索引前的准备（如 codex 载入 title map）
  titleFor?(sid: string): string | null;               // 覆盖标题（codex thread_name）
}
```
